cmake_minimum_required(VERSION 3.16)
project(coordination_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core)
get_filename_component(PACKAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Test-only copies expose orchestration state without changing production interfaces.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/frontend")
file(READ "${PACKAGE_ROOT}/include/frontend/imu_frontend.hpp" IMU_HEADER)
string(REPLACE "private:" "public:" IMU_HEADER "${IMU_HEADER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/frontend/imu_frontend.hpp" "${IMU_HEADER}")
file(READ "${PACKAGE_ROOT}/src/slam_node.cpp" NODE_SOURCE)
string(REPLACE "private:" "public:" NODE_SOURCE "${NODE_SOURCE}")
string(REPLACE "int main(" "int production_main(" NODE_SOURCE "${NODE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/slam_node_testable.cpp" "${NODE_SOURCE}")
add_executable(coordination_test coordination_test.cpp
  "${PACKAGE_ROOT}/src/frontend/imu_frontend.cpp"
  "${PACKAGE_ROOT}/src/frontend/stereo_frontend.cpp")
target_include_directories(coordination_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include"
  "${CMAKE_CURRENT_BINARY_DIR}" "${PACKAGE_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(coordination_test PRIVATE ORB_SLAM3_DEFAULT_VOCABULARY_PATH="/stub/vocabulary")
ament_target_dependencies(coordination_test rclcpp sensor_msgs message_filters cv_bridge)
target_link_libraries(coordination_test Eigen3::Eigen ${OpenCV_LIBS})
target_compile_options(coordination_test PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME coordination COMMAND coordination_test)
